if(BUILD_VERSION STREQUAL "scalar")
    set(SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/scalar")
    if(TARGET_ARCH STREQUAL "riscv")
        add_compile_options( -march=rv64gc )
    elseif(TARGET_ARCH STREQUAL "arm")
        add_compile_options( -march=armv8-a+nosimd -fno-tree-vectorize )
    else()
        add_compile_options( -march=x86-64 -mtune=generic)
    endif()
elseif(BUILD_VERSION STREQUAL "native")
    if(TARGET_ARCH STREQUAL "riscv")
        set(SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/riscv")
        add_compile_options( -march=rv64gcv )
    elseif(TARGET_ARCH STREQUAL "arm")
        set(SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/arm")
        add_compile_options( -march=armv8-a )
    else()
        set(SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/x86")
        if(NOT DEFINED TARGET_ARCH)
            set(TARGET_ARCH "avx2" CACHE STRING "SIMD instruction set level: sse, avx2, avx512")
        endif()
        if(TARGET_ARCH STREQUAL "sse")
            add_compile_options(-march=x86-64 -mtune=generic -msse4.2)
            message(STATUS "Using SSE4.2 instruction set")
        elseif(TARGET_ARCH STREQUAL "avx2")
            add_compile_options(-march=x86-64 -mtune=generic -mavx2 -mfma)
            message(STATUS "Using AVX2 instruction set")
        elseif(TARGET_ARCH STREQUAL "avx512")
            add_compile_options(-march=x86-64 -mtune=generic -mavx512f -mavx512vl -mavx512bw -mavx512dq)
            message(STATUS "Using AVX-512 instruction set")
        else()
            message(FATAL_ERROR "Invalid TARGET_ARCH: ${TARGET_ARCH}. Valid options are: sse, avx2, avx512")
        endif()
    endif()
else()
    set(SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${BUILD_VERSION}")
    message(STATUS "using SRC_DIR: ${SRC_DIR}")
endif()

message(STATUS "using ${SRC_DIR} as source file DIR")

file(GLOB ALL_SRC_FILES "${SRC_DIR}/*.cpp")
if(BUILD_CASES)
    set(SRC_FILES_FOUND "")
    foreach(src_file IN LISTS ALL_SRC_FILES)
        get_filename_component(src_basename ${src_file} NAME_WE)
        if(${src_basename} IN_LIST BUILD_CASES)
            list(APPEND SRC_FILES_FOUND ${src_file})
        endif()
    endforeach()
else()
    set(SRC_FILES_FOUND ${ALL_SRC_FILES})
endif()

set(SRC_FILES_FOUND ${SRC_FILES_FOUND} PARENT_SCOPE)
add_library(static_source STATIC ${SRC_FILES_FOUND})
